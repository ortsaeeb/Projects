// Fair Value Gaps for the Webull Script Editor
// Uses Webull's own names seen in working Webull scripts: define() for settings, ind.sma(), plt().
minPct  = define(0.03, name="Min gap size (% of price)")
dispMul = define(1.25, name="Big candle = body x avg range")

avgRng  = ind.sma(high - low, 20)
body1   = close[1] > open[1] ? close[1] - open[1] : open[1] - close[1]
bigMid  = body1 >= dispMul * avgRng[1]

bullNew = low > high[2] and (low - high[2]) / close * 100 >= minPct and bigMid
bearNew = high < low[2] and (low[2] - high) / close * 100 >= minPct and bigMid

// Remember the latest gap; "live" = 1 until price trades back through it
bullTop  = bullNew ? low : bullTop[1]
bullBot  = bullNew ? high[2] : bullBot[1]
bullLive = bullNew ? 1 : (low <= bullBot[1] ? 0 : bullLive[1])

bearTop  = bearNew ? low[2] : bearTop[1]
bearBot  = bearNew ? high : bearBot[1]
bearLive = bearNew ? 1 : (high >= bearTop[1] ? 0 : bearLive[1])

plt(bullTop, color=bullLive == 1 ? #00C853 : #00000000, name="Bull FVG top")
plt(bullBot, color=bullLive == 1 ? #00C853 : #00000000, name="Bull FVG bottom")
plt(bearTop, color=bearLive == 1 ? #FF1744 : #00000000, name="Bear FVG top")
plt(bearBot, color=bearLive == 1 ? #FF1744 : #00000000, name="Bear FVG bottom")
