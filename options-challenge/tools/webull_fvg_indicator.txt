// Fair Value Gaps for the Webull Script Editor
// Webull scripts can't carry values from bar to bar, so each bar looks back 12 candles (1 hour on 5-min)
// for the newest gap and checks whether price has traded back through it since.
minPct  = define(0.03, name="Min gap size (% of price)")
dispMul = define(1.25, name="Big candle x avg range (0.01 = off)")
gapType = define(1, name="Gap type: 1 = wicks, 2 = bodies")

// Edges used to find a gap: candle wicks (strict) or candle bodies (looser)
bodyHi  = close > open ? close : open
bodyLo  = close > open ? open : close
hiRef   = gapType == 2 ? bodyHi : high
loRef   = gapType == 2 ? bodyLo : low

avgRng  = ind.sma(high - low, 20)
body1   = close[1] > open[1] ? close[1] - open[1] : open[1] - close[1]
bigMid  = body1 >= dispMul * avgRng[1]

bullNew = loRef > hiRef[2] and (loRef - hiRef[2]) / close * 100 >= minPct and bigMid
bearNew = hiRef < loRef[2] and (loRef[2] - hiRef) / close * 100 >= minPct and bigMid

// Newest gap in the last 12 bars: top, bottom, and live = 1 until a wick trades back through its far edge
bullTop  = bullNew ? loRef : (bullNew[1] ? loRef[1] : (bullNew[2] ? loRef[2] : (bullNew[3] ? loRef[3] : (bullNew[4] ? loRef[4] : (bullNew[5] ? loRef[5] : (bullNew[6] ? loRef[6] : (bullNew[7] ? loRef[7] : (bullNew[8] ? loRef[8] : (bullNew[9] ? loRef[9] : (bullNew[10] ? loRef[10] : (bullNew[11] ? loRef[11] : close)))))))))))
bullBot  = bullNew ? hiRef[2] : (bullNew[1] ? hiRef[3] : (bullNew[2] ? hiRef[4] : (bullNew[3] ? hiRef[5] : (bullNew[4] ? hiRef[6] : (bullNew[5] ? hiRef[7] : (bullNew[6] ? hiRef[8] : (bullNew[7] ? hiRef[9] : (bullNew[8] ? hiRef[10] : (bullNew[9] ? hiRef[11] : (bullNew[10] ? hiRef[12] : (bullNew[11] ? hiRef[13] : close)))))))))))
bullLive = bullNew ? 1 : (bullNew[1] ? ((low <= hiRef[3]) ? 0 : 1) : (bullNew[2] ? ((low <= hiRef[4] or low[1] <= hiRef[4]) ? 0 : 1) : (bullNew[3] ? ((low <= hiRef[5] or low[1] <= hiRef[5] or low[2] <= hiRef[5]) ? 0 : 1) : (bullNew[4] ? ((low <= hiRef[6] or low[1] <= hiRef[6] or low[2] <= hiRef[6] or low[3] <= hiRef[6]) ? 0 : 1) : (bullNew[5] ? ((low <= hiRef[7] or low[1] <= hiRef[7] or low[2] <= hiRef[7] or low[3] <= hiRef[7] or low[4] <= hiRef[7]) ? 0 : 1) : (bullNew[6] ? ((low <= hiRef[8] or low[1] <= hiRef[8] or low[2] <= hiRef[8] or low[3] <= hiRef[8] or low[4] <= hiRef[8] or low[5] <= hiRef[8]) ? 0 : 1) : (bullNew[7] ? ((low <= hiRef[9] or low[1] <= hiRef[9] or low[2] <= hiRef[9] or low[3] <= hiRef[9] or low[4] <= hiRef[9] or low[5] <= hiRef[9] or low[6] <= hiRef[9]) ? 0 : 1) : (bullNew[8] ? ((low <= hiRef[10] or low[1] <= hiRef[10] or low[2] <= hiRef[10] or low[3] <= hiRef[10] or low[4] <= hiRef[10] or low[5] <= hiRef[10] or low[6] <= hiRef[10] or low[7] <= hiRef[10]) ? 0 : 1) : (bullNew[9] ? ((low <= hiRef[11] or low[1] <= hiRef[11] or low[2] <= hiRef[11] or low[3] <= hiRef[11] or low[4] <= hiRef[11] or low[5] <= hiRef[11] or low[6] <= hiRef[11] or low[7] <= hiRef[11] or low[8] <= hiRef[11]) ? 0 : 1) : (bullNew[10] ? ((low <= hiRef[12] or low[1] <= hiRef[12] or low[2] <= hiRef[12] or low[3] <= hiRef[12] or low[4] <= hiRef[12] or low[5] <= hiRef[12] or low[6] <= hiRef[12] or low[7] <= hiRef[12] or low[8] <= hiRef[12] or low[9] <= hiRef[12]) ? 0 : 1) : (bullNew[11] ? ((low <= hiRef[13] or low[1] <= hiRef[13] or low[2] <= hiRef[13] or low[3] <= hiRef[13] or low[4] <= hiRef[13] or low[5] <= hiRef[13] or low[6] <= hiRef[13] or low[7] <= hiRef[13] or low[8] <= hiRef[13] or low[9] <= hiRef[13] or low[10] <= hiRef[13]) ? 0 : 1) : 0)))))))))))

bearTop  = bearNew ? loRef[2] : (bearNew[1] ? loRef[3] : (bearNew[2] ? loRef[4] : (bearNew[3] ? loRef[5] : (bearNew[4] ? loRef[6] : (bearNew[5] ? loRef[7] : (bearNew[6] ? loRef[8] : (bearNew[7] ? loRef[9] : (bearNew[8] ? loRef[10] : (bearNew[9] ? loRef[11] : (bearNew[10] ? loRef[12] : (bearNew[11] ? loRef[13] : close)))))))))))
bearBot  = bearNew ? hiRef : (bearNew[1] ? hiRef[1] : (bearNew[2] ? hiRef[2] : (bearNew[3] ? hiRef[3] : (bearNew[4] ? hiRef[4] : (bearNew[5] ? hiRef[5] : (bearNew[6] ? hiRef[6] : (bearNew[7] ? hiRef[7] : (bearNew[8] ? hiRef[8] : (bearNew[9] ? hiRef[9] : (bearNew[10] ? hiRef[10] : (bearNew[11] ? hiRef[11] : close)))))))))))
bearLive = bearNew ? 1 : (bearNew[1] ? ((high >= loRef[3]) ? 0 : 1) : (bearNew[2] ? ((high >= loRef[4] or high[1] >= loRef[4]) ? 0 : 1) : (bearNew[3] ? ((high >= loRef[5] or high[1] >= loRef[5] or high[2] >= loRef[5]) ? 0 : 1) : (bearNew[4] ? ((high >= loRef[6] or high[1] >= loRef[6] or high[2] >= loRef[6] or high[3] >= loRef[6]) ? 0 : 1) : (bearNew[5] ? ((high >= loRef[7] or high[1] >= loRef[7] or high[2] >= loRef[7] or high[3] >= loRef[7] or high[4] >= loRef[7]) ? 0 : 1) : (bearNew[6] ? ((high >= loRef[8] or high[1] >= loRef[8] or high[2] >= loRef[8] or high[3] >= loRef[8] or high[4] >= loRef[8] or high[5] >= loRef[8]) ? 0 : 1) : (bearNew[7] ? ((high >= loRef[9] or high[1] >= loRef[9] or high[2] >= loRef[9] or high[3] >= loRef[9] or high[4] >= loRef[9] or high[5] >= loRef[9] or high[6] >= loRef[9]) ? 0 : 1) : (bearNew[8] ? ((high >= loRef[10] or high[1] >= loRef[10] or high[2] >= loRef[10] or high[3] >= loRef[10] or high[4] >= loRef[10] or high[5] >= loRef[10] or high[6] >= loRef[10] or high[7] >= loRef[10]) ? 0 : 1) : (bearNew[9] ? ((high >= loRef[11] or high[1] >= loRef[11] or high[2] >= loRef[11] or high[3] >= loRef[11] or high[4] >= loRef[11] or high[5] >= loRef[11] or high[6] >= loRef[11] or high[7] >= loRef[11] or high[8] >= loRef[11]) ? 0 : 1) : (bearNew[10] ? ((high >= loRef[12] or high[1] >= loRef[12] or high[2] >= loRef[12] or high[3] >= loRef[12] or high[4] >= loRef[12] or high[5] >= loRef[12] or high[6] >= loRef[12] or high[7] >= loRef[12] or high[8] >= loRef[12] or high[9] >= loRef[12]) ? 0 : 1) : (bearNew[11] ? ((high >= loRef[13] or high[1] >= loRef[13] or high[2] >= loRef[13] or high[3] >= loRef[13] or high[4] >= loRef[13] or high[5] >= loRef[13] or high[6] >= loRef[13] or high[7] >= loRef[13] or high[8] >= loRef[13] or high[9] >= loRef[13] or high[10] >= loRef[13]) ? 0 : 1) : 0)))))))))))

plt(bullTop, color=bullLive == 1 ? #00C853 : #00000000, name="Bull FVG top")
plt(bullBot, color=bullLive == 1 ? #00C853 : #00000000, name="Bull FVG bottom")
plt(bearTop, color=bearLive == 1 ? #FF1744 : #00000000, name="Bear FVG top")
plt(bearBot, color=bearLive == 1 ? #FF1744 : #00000000, name="Bear FVG bottom")
