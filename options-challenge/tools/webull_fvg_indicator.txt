// Fair Value Gaps for the Webull Script Editor
study(title="Fair Value Gaps", shorttitle="FVG", overlay=true)

// Settings: edit the numbers here
minPct  = 0.03     // minimum gap size, % of price
dispOn  = true     // only gaps whose middle candle is big
dispMul = 1.25     // big = body at least this x the 20-bar average range

rngSum  = (high - low) + (high[1] - low[1]) + (high[2] - low[2]) + (high[3] - low[3]) + (high[4] - low[4]) + (high[5] - low[5]) + (high[6] - low[6]) + (high[7] - low[7]) + (high[8] - low[8]) + (high[9] - low[9]) + (high[10] - low[10]) + (high[11] - low[11]) + (high[12] - low[12]) + (high[13] - low[13]) + (high[14] - low[14]) + (high[15] - low[15]) + (high[16] - low[16]) + (high[17] - low[17]) + (high[18] - low[18]) + (high[19] - low[19])
avgRng  = rngSum / 20
body1   = close[1] > open[1] ? close[1] - open[1] : open[1] - close[1]
bigMid  = body1 >= dispMul * avgRng[1]
okDisp  = dispOn ? bigMid : true

bullNew = low > high[2] and (low - high[2]) / close * 100 >= minPct and okDisp
bearNew = high < low[2] and (low[2] - high) / close * 100 >= minPct and okDisp

var float bullTop = na
var float bullBot = na
var float bearTop = na
var float bearBot = na

bullFilled = not na(bullBot) and low <= bullBot
bearFilled = not na(bearTop) and high >= bearTop

if bullFilled
    bullTop := na
    bullBot := na
if bullNew
    bullTop := low
    bullBot := high[2]

if bearFilled
    bearTop := na
    bearBot := na
if bearNew
    bearTop := low[2]
    bearBot := high

p1 = plot(bullTop, title="Bull FVG top", color=color.green, style=plot.style_linebr)
p2 = plot(bullBot, title="Bull FVG bottom", color=color.green, style=plot.style_linebr)
fill(p1, p2, color=color.new(color.green, 80))

p3 = plot(bearTop, title="Bear FVG top", color=color.red, style=plot.style_linebr)
p4 = plot(bearBot, title="Bear FVG bottom", color=color.red, style=plot.style_linebr)
fill(p3, p4, color=color.new(color.red, 80))

plotshape(bullNew, title="New bull FVG", style=shape.triangleup, location=location.belowbar, color=color.green, size=size.tiny)
plotshape(bearNew, title="New bear FVG", style=shape.triangledown, location=location.abovebar, color=color.red, size=size.tiny)
